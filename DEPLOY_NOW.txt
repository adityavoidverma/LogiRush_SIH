📍 STEP 1: RUN PRE-FLIGHT CHECK (1 min)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    cd /Users/user/Downloads/LogiRush
    ./pre-deploy-check.sh

If errors appear, fix them before continuing.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📍 STEP 2: PUSH TO GITHUB (3 min)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    # If not already a git repo:
    git init
    git add .
    git commit -m "Ready for deployment"
    
    # Create repo at github.com/new, then:
    git remote add origin https://github.com/YOUR_USERNAME/LogiRush.git
    git branch -M main
    git push -u origin main

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📍 STEP 3: CREATE DATABASE (5 min)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    1. Visit: https://console.neon.tech/app/projects
    2. Click "New Project"
    3. Name: logirush-db
    4. Copy connection string (starts with postgresql://)
    5. SAVE IT! You'll need it in 2 minutes

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📍 STEP 4: DEPLOY BACKEND TO RENDER (10 min)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    OPTION A - One Click (Recommended):
    
    1. Visit: https://render.com/deploy?repo=https://github.com/YOUR_USERNAME/LogiRush
    2. Connect GitHub account
    3. Fill environment variables:
       
       DATABASE_URL = [paste Neon connection string]
       CORS_ORIGINS = *
       PORT = 10000
       NER_DISABLE_DEMO_SEED = [leave empty]
    
    4. Click "Apply"
    5. Wait ~8 minutes for build
    6. Copy your URL: https://ner-logistics-backend-XXXX.onrender.com
    
    OPTION B - Manual:
    See QUICK_DEPLOY.md for manual steps

    ✅ TEST: Visit https://YOUR-BACKEND.onrender.com/health
            Should see: {"status":"ok"}

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📍 STEP 5: DEPLOY FRONTEND TO VERCEL (5 min)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    OPTION A - One Click (Recommended):
    
    1. Visit: https://vercel.com/new
    2. Import your GitHub repo
    3. CRITICAL SETTINGS:
       
       ⚠️  Root Directory: routeOptimiserFrontend  ⚠️
       
       (Click "Edit" to change from "/" to "routeOptimiserFrontend")
       
    4. Add Environment Variable:
       
       VITE_API_BASE_URL = https://your-backend.onrender.com
       
       (Use the URL from Step 4, NO TRAILING SLASH)
    
    5. Click "Deploy"
    6. Wait ~3 minutes
    7. Copy your URL: https://logirush-XXXX.vercel.app

    ✅ TEST: Open your Vercel URL in browser
            Should see LogiRush dashboard

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📍 STEP 6: UPDATE CORS (2 min)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    Now backend needs to allow frontend requests:
    
    1. Go to: https://dashboard.render.com
    2. Click your backend service
    3. Go to "Environment" tab
    4. Find CORS_ORIGINS
    5. Change from "*" to: https://logirush-XXXX.vercel.app
       (Your actual Vercel URL from Step 5)
    6. Click "Save Changes"
    7. Wait ~2 mins for redeploy

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📍 STEP 7: TEST DEPLOYMENT (3 min)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    Automated:
    ./verify-deployment.sh
    
    Manual:
    1. Open your Vercel URL
    2. Sign in:
       Username: controller
       Password: control123
    3. Check dashboard loads
    4. Try creating a route

📍 STEP 8: SETUP MOBILE APP (Optional, 5 min)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    cd nerFieldApp
    
    echo "EXPO_PUBLIC_API_BASE_URL=https://YOUR-BACKEND.onrender.com" > .env
    
    npm install
    npx expo start
